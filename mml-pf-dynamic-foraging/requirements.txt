# unpinned installs will silently change numerical output between runs.

numpy
scipy
pandas

# Required on COMPUTE NODES too, even though no compute job calls them.
# `import hpc_block_cv` -> `import mmlpf_vs_psytrack_cv`, which imports
# psytrack and matplotlib at module level. Omit either from the container and
# every array task dies at import with a ModuleNotFoundError. Moving those two
# imports inside the functions that use them would let the compute image drop
# both, and is worth doing if container size ever matters.
psytrack
matplotlib
